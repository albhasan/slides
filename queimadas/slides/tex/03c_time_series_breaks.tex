\begin{frame}
\frametitle{Breaks For Additive Seasonal and Trend - BFAST}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{itemize}
        \item Additive decomposition model that iteratively fits a linear trend and a seasonal model.
        \item $Y_{t}$ is the observed data at time \textit{t}.
        \item $T_{t}$ is the trend component.
        \item $S_{t}$ is the season component.
        \item $e_{t}$ is the remaining variation in the data beyond the seasonal and trend components.
        \item \cite{Verbesselt:2010}.
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
    \centering
    $Y_{t} = T_{t} + S_{t} + e_{t}$
    \end{column}
  \end{columns}
\end{frame}
